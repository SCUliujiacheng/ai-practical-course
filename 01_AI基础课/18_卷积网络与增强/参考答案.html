2. 修改：移到左边

写一个零填充左平移函数，用非对称小图核对，不得环绕边缘。

def shift_left(batch):
    out = torch.zeros_like(batch)
    out[:, :, :, :-1] = batch[:, :, :, 1:]
    return out
a = torch.arange(8.0).reshape(1, 1, 2, 4)
b = shift_left(a)
assert torch.equal(b[0, 0, 0], torch.tensor([1.0, 2.0, 3.0, 0.0]))
assert b[:, :, :, -1].sum() == 0
print(b)
tensor([[[[1., 2., 3., 0.],
