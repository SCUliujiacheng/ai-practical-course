new_docs = {**documents, 'D7': '会议室预约必须填写参会人数。会议结束后应关闭投影仪。'}
v = TfidfVectorizer(analyzer='char', ngram_range=(2, 4))

score = (m @ v.transform(['会议室参会人数']).T).toarray().ravel()
