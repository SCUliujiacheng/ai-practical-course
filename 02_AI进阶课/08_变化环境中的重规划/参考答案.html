saved_events=events
try:
    events=[]
    unchanged=[execute(p) for p in ["fixed","repair_blocked","rolling"]]
    assert all(r["success"] and r["time"]==2 for r in unchanged)
    assert all(r["changes"]==0 for r in unchanged)
finally: events=saved_events
